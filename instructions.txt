CONTROL SIGNALS 1
========================================================================================================================================================
Bit	Signal		    Worth	Hex value		Wire goes to
0	RAM_READ	    1	    0x00000001		RAM read (tri-states onto bus)
1	RAM_WRITE	    2	    0x00000002		RAM write
2	A_LOAD		    4	    0x00000004		A register load
3	A_OUT		    8	    0x00000008		A's tri-state onto bus
4	B_LOAD		    1	    0x00000010		B register load
5	ALU_OUT		    2	    0x00000020		ALU's tri-state onto bus
6	SUB		        4	    0x00000040		ALU subtract
7	VAL_OUT		    8	    0x00000080		value field's tri-state onto bus
8	PC_INC		    1	    0x00000100		PC count up
9	PC_LOAD		    2	    0x00000200		PC load (jump)
10	STEP_RESET	    4	    0x00000400		step counter reset
11	FLAGS_LOAD	    8	    0x00000800		save zero/carry flags
12	JUMP_IF_ZERO	1	    0x00001000		conditional jump logic
13	JUMP_IF_CARRY	2	    0x00002000		conditional jump logic
14	OUT_LOAD	    4	    00004000		output register load
15	HALT		    8	    0x00008000		stops the clock
16	IN_EN			1		0x00010000		Input tri-state onto bus
17  B_OUT			2		0x00020000


========================================================================================================================================================
REGISTERS
========================================================================================================================================================
A 	 - Main register
B  	 - Sub register
OUT  - Output register
PC 	 - Program counter
STEP - Step counter


========================================================================================================================================================
FLAGS
========================================================================================================================================================
Z - Zero flag
C - Carry flag


========================================================================================================================================================
EXTRA LOGIC
========================================================================================================================================================
Port enable    = IN_EN AND decoder line (decoder on addr bits 1-0)
Zero-fill      = VAL_OUT OR (IN_EN AND line 2)
JUMP           = PC_LOAD OR (JUMP_IF_ZERO AND Z) OR (JUMP_IF_CARRY AND C)
PC LOAD        = PC_INC OR JUMP OR RESET
Step clear     = STEP_RESET OR RESET
CPU clock      = CLOCK AND (NOT HALT OR RESET)
